Contexto
Al entrar en la Pestaña Lectura y al hacer clic en la subpestaña Voces e Idiomas (Ajustes), wxPython congela la interfaz ~2 segundos porque carga JSONs pesados y reconstruye listas completas en el hilo principal (main thread). NVDA pierde el foco durante ese bloqueo.

La solución es desplazar toda la I/O de disco y el llenado inicial de controles a un threading.Thread, y usar wx.CallAfter para actualizar los controles de vuelta en el hilo principal de forma segura.

def _cargar_voces_hilo(self):
    """Carga JSONs de voces fuera del hilo principal para evitar bloqueo de foco."""
    self._cargando_voces = True
    # --- toda la lógica actual de lectura de JSON y construcción de lista ---
    # Al final, en lugar de poblar el combo directamente:
    wx.CallAfter(self._aplicar_voces_en_ui, voces_formateadas, seleccion_previa)
    self._cargando_voces = False

def _aplicar_voces_en_ui(self, voces_formateadas, seleccion_previa):
    """Llena el combo de voces y reactiva controles. Siempre en hilo principal."""
    self.combo_voz.Clear()
    for nombre, datos in voces_formateadas:
        idx = self.combo_voz.Append(nombre)
        self.combo_voz.SetClientData(idx, datos)
    self.combo_voz.SetSelection(seleccion_previa)
    self.al_cambiar_voz(None)
    self._habilitar_controles_voz()
    # Situar foco en el combo para que NVDA lo anuncie inmediatamente
    wx.CallAfter(self.combo_voz.SetFocus)

def _deshabilitar_controles_voz(self):
    self.combo_voz.Disable()

def _habilitar_controles_voz(self):
    self.combo_voz.Enable()
